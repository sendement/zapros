# zapros' reference of functions, operators and keywords (Ctrl+click).
#
#   @ names (comma-separated) | dialects (* or pg ch ora lite) | f / o / k
#   sig: how it's written
#   ru: / en: what it does (one line each, may repeat)
#   ex: an example (each line its own ex:)
#
# Entries for one name are looked up first by the dialect, then "*".

# ---------------------------------------------------------- aggregates

@ count | * | f
sig: count(*) | count(expr) | count(DISTINCT expr)
ru: Число строк (*) или не-NULL значений выражения; с DISTINCT — число разных значений.
en: The number of rows (*) or of the expression's non-NULL values; with DISTINCT, of different values.
ex: SELECT customer_id, count(*) AS orders, count(DISTINCT status) AS statuses
ex:   FROM orders
ex:  GROUP BY customer_id;

@ sum | * | f
sig: sum(expr) | sum(DISTINCT expr)
ru: Сумма значений группы; NULL не считаются, у пустой группы — NULL.
en: The sum of the group's values; NULLs are skipped, an empty group gives NULL.
ex: SELECT customer_id, sum(total) AS spent
ex:   FROM orders
ex:  GROUP BY customer_id;

@ avg | * | f
sig: avg(expr)
ru: Среднее арифметическое группы (NULL не учитываются).
en: The group's arithmetic mean (NULLs left out).
ex: SELECT avg(total) FROM orders WHERE created_at >= DATE '2026-01-01';

@ min, max | * | f
sig: min(expr) / max(expr)
ru: Наименьшее / наибольшее значение группы: числа, строки, даты.
en: The group's least / greatest value: numbers, strings, dates.
ex: SELECT min(created_at) AS first_order, max(total) AS biggest FROM orders;

@ string_agg | pg | f
sig: string_agg(expr, separator [ORDER BY …])
ru: Склеивает значения группы в одну строку через разделитель.
en: Joins the group's values into one string with a separator.
ex: SELECT customer_id, string_agg(product, ', ' ORDER BY product)
ex:   FROM order_items GROUP BY customer_id;

@ array_agg | pg | f
sig: array_agg(expr [ORDER BY …])
ru: Собирает значения группы в массив.
en: Gathers the group's values into an array.
ex: SELECT customer_id, array_agg(id ORDER BY created_at) AS order_ids FROM orders GROUP BY customer_id;

@ listagg | ora | f
sig: LISTAGG(expr, separator) WITHIN GROUP (ORDER BY …)
ru: Склеивает значения группы в строку через разделитель, в заданном порядке.
en: Joins the group's values into a string with a separator, in the order given.
ex: SELECT customer_id, LISTAGG(product, ', ') WITHIN GROUP (ORDER BY product) AS products
ex:   FROM order_items GROUP BY customer_id;

@ group_concat | lite | f
sig: group_concat(expr [, separator])
ru: Склеивает значения группы в строку (по умолчанию через запятую).
en: Joins the group's values into a string (a comma by default).
ex: SELECT customer_id, group_concat(product, '; ') FROM order_items GROUP BY customer_id;

@ groupArray | ch | f
sig: groupArray([max_size])(x)
ru: Собирает значения группы в массив (с ограничением размера, если задано).
en: Gathers the group's values into an array (up to max_size, if given).
ex: SELECT customer_id, groupArray(10)(id) AS last_ids FROM orders GROUP BY customer_id;

@ uniq, uniqExact | ch | f
sig: uniq(x) | uniqExact(x)
ru: Число разных значений: uniq — приближённо и быстро, uniqExact — точно.
en: The number of distinct values: uniq approximately and fast, uniqExact exactly.
ex: SELECT toDate(ts) AS day, uniq(user_id) AS users FROM visits GROUP BY day;

@ countIf, sumIf, avgIf, minIf, maxIf, anyIf | ch | f
sig: countIf(cond) | sumIf(x, cond) | avgIf(x, cond)
ru: Агрегат только по строкам, где условие истинно (комбинатор -If).
en: An aggregate over the rows where the condition holds (the -If combinator).
ex: SELECT countIf(status = 'paid') AS paid, sumIf(total, status = 'paid') AS paid_sum FROM orders;

@ argMax, argMin | ch | f
sig: argMax(arg, val) / argMin(arg, val)
ru: Значение arg в строке с наибольшим / наименьшим val.
en: The arg of the row with the greatest / least val.
ex: SELECT customer_id, argMax(status, updated_at) AS last_status FROM orders GROUP BY customer_id;

@ any, anyLast | ch | f
sig: any(x) / anyLast(x)
ru: Какое-нибудь (первое / последнее встреченное) значение группы.
en: Some (the first / last met) value of the group.
ex: SELECT customer_id, any(name) FROM orders GROUP BY customer_id;

@ quantile, median | ch | f
sig: quantile(level)(x) | median(x)
ru: Приближённый квантиль уровня 0…1; median = quantile(0.5).
en: An approximate quantile of level 0…1; median = quantile(0.5).
ex: SELECT quantile(0.95)(duration_ms) AS p95 FROM requests;

@ percentile_cont | pg ora | f
sig: percentile_cont(fraction) WITHIN GROUP (ORDER BY expr)
ru: Непрерывный перцентиль (с интерполяцией); 0.5 — медиана.
en: A continuous percentile (interpolated); 0.5 is the median.
ex: SELECT percentile_cont(0.5) WITHIN GROUP (ORDER BY total) AS median FROM orders;

@ total | lite | f
sig: total(expr)
ru: Сумма как число с плавающей точкой; у пустой группы — 0.0, а не NULL.
en: The sum as a float; 0.0 for an empty group, not NULL.
ex: SELECT total(amount) FROM payments WHERE 0;

# ----------------------------------------------------------- window

@ row_number | * | f
sig: row_number() OVER ([PARTITION BY …] ORDER BY …)
ru: Номер строки внутри окна: 1, 2, 3… без повторов.
en: The row's number in its window: 1, 2, 3… with no repeats.
ex: SELECT * FROM (
ex:   SELECT o.*, row_number() OVER (PARTITION BY customer_id ORDER BY created_at DESC) AS rn
ex:     FROM orders o) t
ex:  WHERE rn = 1;  -- each customer's last order

@ rank, dense_rank | * | f
sig: rank() / dense_rank() OVER (… ORDER BY …)
ru: Место в окне; равные делят место. rank пропускает номера после равных (1,1,3), dense_rank — нет (1,1,2).
en: The place in the window; equals share it. rank skips numbers after ties (1,1,3), dense_rank doesn't (1,1,2).
ex: SELECT name, score, dense_rank() OVER (ORDER BY score DESC) AS place FROM players;

@ lag, lead | pg ora lite | f
sig: lag(expr [, offset [, default]]) OVER (…) / lead(…)
ru: Значение из предыдущей (lag) или следующей (lead) строки окна.
en: The value from the previous (lag) or next (lead) row of the window.
ex: SELECT day, revenue, revenue - lag(revenue) OVER (ORDER BY day) AS change FROM daily;

@ first_value, last_value | * | f
sig: first_value(expr) OVER (…) / last_value(expr) OVER (… ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING)
ru: Первое / последнее значение окна. Для last_value задайте рамку до конца окна, иначе это текущая строка.
en: The window's first / last value. For last_value set the frame to the window's end, else it's the current row.
ex: SELECT id, first_value(status) OVER (PARTITION BY customer_id ORDER BY created_at) FROM orders;

@ ntile | * | f
sig: ntile(n) OVER (ORDER BY …)
ru: Делит строки окна на n примерно равных групп и даёт номер группы.
en: Splits the window's rows into n near-equal groups, giving the group's number.
ex: SELECT customer_id, spent, ntile(4) OVER (ORDER BY spent DESC) AS quartile FROM totals;

@ over | * | k
sig: агрегат(…) OVER ([PARTITION BY …] [ORDER BY …] [ROWS | RANGE …])
ru: Делает агрегат или ранжирующую функцию оконной: она считается по «окну» строк, а строки не сворачиваются.
ru: PARTITION BY — на какие окна делить, ORDER BY — порядок внутри, ROWS/RANGE — рамка.
en: Makes an aggregate or ranking function a window one: computed over a window of rows, the rows aren't collapsed.
en: PARTITION BY splits into windows, ORDER BY orders inside, ROWS/RANGE set the frame.
ex: SELECT id, total,
ex:        sum(total) OVER (PARTITION BY customer_id ORDER BY created_at
ex:                         ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS running
ex:   FROM orders;

@ partition | * | k
sig: OVER (PARTITION BY expr, …)
ru: Внутри OVER: делит строки на независимые окна — по каждому окну функция считается отдельно.
en: Inside OVER: splits the rows into independent windows, the function computed for each.
ex: SELECT id, avg(total) OVER (PARTITION BY customer_id) AS customer_avg FROM orders;

@ runningDifference | ch | f
sig: runningDifference(x)
ru: Разница со значением предыдущей строки блока (порядок — как пришли строки; сортируйте в подзапросе).
en: The difference with the previous row's value in the block (in arrival order; sort in a subquery).
ex: SELECT day, runningDifference(users) FROM (SELECT day, users FROM daily ORDER BY day);

# ----------------------------------------------------------- null & logic

@ coalesce | * | f
sig: coalesce(a, b, …)
ru: Первое значение из списка, которое не NULL; если все NULL — NULL.
en: The first value of the list that isn't NULL; NULL if all are.
ex: SELECT coalesce(phone, email, 'нет контакта') AS contact FROM customers;

@ nullif | * | f
sig: nullif(a, b)
ru: NULL, если a = b, иначе a. Удобно против деления на ноль.
en: NULL when a = b, else a. Handy against division by zero.
ex: SELECT paid / nullif(total, 0) AS share FROM orders;

@ nvl | ora | f
sig: NVL(expr, default)
ru: default, если expr — NULL, иначе expr (как coalesce для двух).
en: default when expr is NULL, else expr (coalesce of two).
ex: SELECT NVL(discount, 0) FROM orders;

@ nvl2 | ora | f
sig: NVL2(expr, if_not_null, if_null)
ru: Второй аргумент, если expr не NULL, иначе третий.
en: The second argument when expr isn't NULL, else the third.
ex: SELECT NVL2(shipped_at, 'отправлен', 'ждёт') FROM orders;

@ ifnull | lite ch | f
sig: ifnull(x, default)
ru: default, если x — NULL, иначе x.
en: default when x is NULL, else x.
ex: SELECT ifnull(note, '') FROM orders;

@ decode | ora | f
sig: DECODE(expr, search1, result1 [, search2, result2 …] [, default])
ru: Сравнивает expr с вариантами и даёт результат первого совпавшего (NULL = NULL здесь равны).
en: Compares expr with the options and gives the first match's result (NULL equals NULL here).
ex: SELECT DECODE(status, 'P', 'оплачен', 'C', 'отменён', 'новый') FROM orders;

@ if | ch | f
sig: if(cond, then, else)
ru: then, если условие истинно, иначе else.
en: then when the condition holds, else else.
ex: SELECT if(total > 1000, 'big', 'small') AS kind FROM orders;

@ multiIf | ch | f
sig: multiIf(cond1, then1, cond2, then2, …, else)
ru: Как CASE WHEN: результат первого истинного условия, иначе последний аргумент.
en: Like CASE WHEN: the first true condition's result, else the last argument.
ex: SELECT multiIf(total > 1000, 'big', total > 100, 'medium', 'small') FROM orders;

@ iif | lite | f
sig: iif(cond, then, else)
ru: then, если условие истинно, иначе else (SQLite 3.32+).
en: then when the condition holds, else else (SQLite 3.32+).
ex: SELECT iif(total > 100, 'big', 'small') FROM orders;

@ greatest, least | pg ora ch | f
sig: greatest(a, b, …) / least(a, b, …)
ru: Наибольшее / наименьшее из перечисленных значений (не агрегат: по одной строке).
en: The greatest / least of the values listed (not an aggregate: within a row).
ex: SELECT greatest(price, min_price) FROM products;

@ case | * | k
sig: CASE WHEN cond THEN res … [ELSE res] END | CASE expr WHEN value THEN res … END
ru: Условное выражение: результат первой подошедшей ветки, иначе ELSE (без ELSE — NULL).
en: A conditional expression: the first matching branch's result, else ELSE (without it, NULL).
ex: SELECT CASE WHEN total > 1000 THEN 'big'
ex:             WHEN total > 100 THEN 'medium'
ex:             ELSE 'small' END AS kind
ex:   FROM orders;

@ is | * | k
sig: x IS [NOT] NULL | x IS [NOT] DISTINCT FROM y | x IS [NOT] TRUE
ru: Проверка на NULL (= NULL всегда даёт NULL, нужен IS NULL). IS DISTINCT FROM сравнивает, считая NULL обычным значением.
en: A test for NULL (= NULL is always NULL, IS NULL is needed). IS DISTINCT FROM compares, NULL taken as a plain value.
ex: SELECT * FROM orders WHERE shipped_at IS NULL;
ex: SELECT * FROM a JOIN b ON a.code IS NOT DISTINCT FROM b.code;

@ in | * | k
sig: x [NOT] IN (v1, v2, …) | x [NOT] IN (SELECT …)
ru: Входит ли значение в список или в результат подзапроса. Осторожно: NOT IN с NULL в списке не даёт ни одной строки.
en: Whether the value is in the list or the subquery's result. Careful: NOT IN with a NULL in the list gives no rows.
ex: SELECT * FROM orders WHERE status IN ('new', 'paid');
ex: SELECT * FROM customers WHERE id NOT IN (SELECT customer_id FROM orders WHERE customer_id IS NOT NULL);

@ between | * | k
sig: x [NOT] BETWEEN a AND b
ru: a <= x AND x <= b — обе границы включены. Для дат-времени удобнее >= и <.
en: a <= x AND x <= b, both ends in. For timestamps >= and < are handier.
ex: SELECT * FROM orders WHERE total BETWEEN 100 AND 500;

@ like | * | k
sig: x [NOT] LIKE pattern [ESCAPE c]
ru: Сравнение с шаблоном: % — любая строка, _ — один символ. В PostgreSQL и ClickHouse чувствителен к регистру.
en: A pattern match: % any string, _ one character. Case-sensitive in PostgreSQL and ClickHouse.
ex: SELECT * FROM customers WHERE email LIKE '%@example.com';

@ ilike | pg ch | k
sig: x [NOT] ILIKE pattern
ru: LIKE без учёта регистра.
en: LIKE ignoring case.
ex: SELECT * FROM customers WHERE name ILIKE 'иван%';

@ exists | * | k
sig: [NOT] EXISTS (SELECT …)
ru: Истина, если подзапрос вернул хотя бы одну строку (что в SELECT — не важно).
en: True when the subquery gives at least one row (what it selects doesn't matter).
ex: SELECT * FROM customers c
ex:  WHERE EXISTS (SELECT 1 FROM orders o WHERE o.customer_id = c.id);

@ any, some, all | pg ora | k
sig: x op ANY (SELECT … | array) / x op ALL (…)
ru: Сравнение с каждым значением подзапроса или массива: ANY — хотя бы одно подходит, ALL — все.
en: Comparison with each value of a subquery or array: ANY at least one fits, ALL every one.
ex: SELECT * FROM products WHERE price > ALL (SELECT price FROM products WHERE category = 'eco');

@ distinct | * | k
sig: SELECT DISTINCT … | count(DISTINCT x) | DISTINCT ON (expr) (PostgreSQL)
ru: Убирает повторяющиеся строки результата. DISTINCT ON (PostgreSQL) оставляет первую строку на каждое значение выражения (по ORDER BY).
en: Drops repeated result rows. DISTINCT ON (PostgreSQL) keeps the first row per value of the expression (by ORDER BY).
ex: SELECT DISTINCT ON (customer_id) * FROM orders ORDER BY customer_id, created_at DESC;

# ----------------------------------------------------------- joins & sets

@ join, inner | * | k
sig: a [INNER] JOIN b ON cond | a JOIN b USING (col)
ru: Строки, у которых есть пара в обеих таблицах по условию.
en: Rows that have a match in both tables by the condition.
ex: SELECT o.id, c.name FROM orders o JOIN customers c ON c.id = o.customer_id;

@ left | * | k
sig: a LEFT [OUTER] JOIN b ON cond
ru: Все строки a; поля b — NULL, где пары не нашлось.
en: Every row of a; b's columns NULL where there's no match.
ex: SELECT c.name, count(o.id) FROM customers c LEFT JOIN orders o ON o.customer_id = c.id GROUP BY c.name;

@ right | * | k
sig: a RIGHT [OUTER] JOIN b ON cond
ru: Все строки b; поля a — NULL, где пары не нашлось. Обычно пишут LEFT JOIN, поменяв таблицы.
en: Every row of b; a's columns NULL where there's no match. Usually written as LEFT JOIN with the tables swapped.
ex: SELECT * FROM orders o RIGHT JOIN customers c ON c.id = o.customer_id;

@ full | * | k
sig: a FULL [OUTER] JOIN b ON cond
ru: Все строки обеих таблиц; там, где пары нет, недостающая сторона — NULL.
en: Every row of both tables; where there's no match, the missing side is NULL.
ex: SELECT coalesce(a.day, b.day) AS day, a.sales, b.costs FROM a FULL JOIN b ON a.day = b.day;

@ cross | * | k
sig: a CROSS JOIN b
ru: Каждая строка a с каждой строкой b (декартово произведение).
en: Each row of a with each row of b (the Cartesian product).
ex: SELECT d.day, p.id FROM days d CROSS JOIN products p;

@ lateral | pg | k
sig: FROM a, LATERAL (SELECT … WHERE … = a.x) b | JOIN LATERAL (…) b ON true
ru: Подзапрос во FROM, который видит столбцы таблиц левее — «для каждой строки».
en: A FROM subquery that sees the columns of tables to its left: "for each row".
ex: SELECT c.name, last.total
ex:   FROM customers c
ex:   JOIN LATERAL (SELECT total FROM orders o WHERE o.customer_id = c.id
ex:                  ORDER BY created_at DESC LIMIT 1) last ON true;

@ union | * | k
sig: SELECT … UNION [ALL] SELECT …
ru: Склеивает результаты запросов. UNION убирает повторы (медленнее), UNION ALL — оставляет.
en: Stacks the queries' results. UNION drops repeats (slower), UNION ALL keeps them.
ex: SELECT email FROM customers UNION SELECT email FROM subscribers;

@ intersect, except, minus | * | k
sig: SELECT … INTERSECT SELECT … | SELECT … EXCEPT (Oracle: MINUS) SELECT …
ru: INTERSECT — строки, что есть в обоих результатах; EXCEPT/MINUS — что есть в первом, но нет во втором.
en: INTERSECT: rows in both results; EXCEPT/MINUS: rows in the first but not the second.
ex: SELECT customer_id FROM orders EXCEPT SELECT id FROM customers;

@ with | * | k
sig: WITH name [(cols)] AS (SELECT …) [, …] SELECT … | WITH RECURSIVE …
ru: Именованные подзапросы (CTE) перед основным запросом. RECURSIVE — CTE, ссылающийся на себя (деревья, ряды).
en: Named subqueries (CTEs) before the main query. RECURSIVE: a CTE that refers to itself (trees, series).
ex: WITH big AS (SELECT * FROM orders WHERE total > 1000)
ex: SELECT customer_id, count(*) FROM big GROUP BY customer_id;

@ arrayJoin | ch | f
sig: arrayJoin(arr) | … ARRAY JOIN arr AS x
ru: Разворачивает массив в строки: по строке на элемент.
en: Unfolds an array into rows: a row per element.
ex: SELECT id, arrayJoin(tags) AS tag FROM posts;

@ unnest | pg | f
sig: unnest(array [, …])
ru: Разворачивает массив в строки.
en: Unfolds an array into rows.
ex: SELECT id, unnest(tags) AS tag FROM posts;

@ generate_series | pg | f
sig: generate_series(start, stop [, step])
ru: Ряд чисел или дат от start до stop с шагом — как таблица.
en: A series of numbers or dates from start to stop by step, as a table.
ex: SELECT d::date FROM generate_series(DATE '2026-01-01', DATE '2026-01-31', INTERVAL '1 day') d;

# ----------------------------------------------------------- strings

@ length | pg ora lite | f
sig: length(s)
ru: Длина строки в символах (Oracle: LENGTHB — в байтах).
en: The string's length in characters (Oracle: LENGTHB in bytes).
ex: SELECT name FROM customers WHERE length(name) > 30;

@ length | ch | f
sig: length(s) | lengthUTF8(s)
ru: length — длина строки в БАЙТАХ (и длина массива); для символов — lengthUTF8.
en: length is the string's length in BYTES (and an array's length); characters: lengthUTF8.
ex: SELECT length('привет'), lengthUTF8('привет');  -- 12, 6

@ lower, upper | * | f
sig: lower(s) / upper(s)
ru: Строка в нижнем / верхнем регистре. В ClickHouse для кириллицы — lowerUTF8 / upperUTF8.
en: The string in lower / upper case. In ClickHouse, lowerUTF8 / upperUTF8 for non-Latin text.
ex: SELECT * FROM customers WHERE lower(email) = lower(:email);

@ initcap | pg ora | f
sig: initcap(s)
ru: Каждое слово с заглавной буквы.
en: Each word capitalised.
ex: SELECT initcap('иван петров');  -- Иван Петров

@ trim, ltrim, rtrim | * | f
sig: trim([BOTH | LEADING | TRAILING] [chars FROM] s) | ltrim(s) | rtrim(s)
ru: Убирает пробелы (или указанные символы) по краям строки.
en: Strips spaces (or the characters given) from the string's ends.
ex: SELECT trim(both '0' from '000123000'), rtrim(note) FROM orders;

@ substr, substring | * | f
sig: substr(s, from [, count]) | substring(s FROM from FOR count)
ru: Часть строки с позиции from (с 1) длиной count. В Oracle отрицательное from — от конца.
en: Part of the string from position from (1-based), count long. In Oracle a negative from counts from the end.
ex: SELECT substr(phone, 1, 3) AS code FROM customers;

@ replace | * | f
sig: replace(s, what, with)
ru: Заменяет все вхождения подстроки. В ClickHouse — replaceAll (replaceOne — первое).
en: Replaces every occurrence of a substring. In ClickHouse, replaceAll (replaceOne: the first).
ex: SELECT replace(phone, '-', '') FROM customers;

@ replaceAll, replaceOne | ch | f
sig: replaceAll(s, what, with) / replaceOne(…)
ru: Замена всех / первого вхождения подстроки.
en: Replaces every / the first occurrence of a substring.
ex: SELECT replaceAll(url, 'http://', 'https://') FROM links;

@ concat | * | f
sig: concat(a, b, …)
ru: Склейка строк. В PostgreSQL и Oracle NULL-ы пропускаются (в отличие от ||).
en: Joins strings. In PostgreSQL and Oracle NULLs are skipped (unlike ||).
ex: SELECT concat(first_name, ' ', last_name) FROM customers;

@ position, strpos, instr | * | f
sig: position(sub IN s) | strpos(s, sub) | instr(s, sub [, from [, nth]])
ru: Позиция подстроки в строке (с 1); 0, если нет. Oracle: INSTR, ClickHouse: position(s, sub).
en: The substring's position in the string (1-based); 0 when absent. Oracle: INSTR, ClickHouse: position(s, sub).
ex: SELECT substr(email, instr(email, '@') + 1) AS domain FROM customers;

@ lpad, rpad | pg ora ch | f
sig: lpad(s, length [, fill]) / rpad(…)
ru: Дополняет строку слева / справа до длины (по умолчанию пробелами).
en: Pads the string on the left / right to the length (spaces by default).
ex: SELECT lpad(id::text, 8, '0') FROM orders;

@ split_part | pg | f
sig: split_part(s, separator, n)
ru: n-я часть строки, разделённой разделителем (с 1; отрицательное — с конца).
en: The n-th part of the string split by the separator (1-based; negative from the end).
ex: SELECT split_part(email, '@', 2) AS domain FROM customers;

@ splitByChar, splitByString | ch | f
sig: splitByChar(sep, s) / splitByString(sep, s)
ru: Массив частей строки по разделителю.
en: An array of the string's parts by the separator.
ex: SELECT splitByChar('@', email)[2] AS domain FROM customers;

ru: Поле JSON по ключу или элемент по индексу — результат JSON.
en: A JSON field by key or element by index, as JSON.
ex: SELECT payload -> 'user' FROM events;

@ ->> | pg lite | o
sig: json ->> 'key'
ru: Поле JSON по ключу — результат текстом.
en: A JSON field by key, as text.
ex: SELECT payload ->> 'email' FROM events WHERE payload ->> 'type' = 'signup';

@ #>, #>> | pg | o
sig: json #> '{a,b}' / json #>> '{a,b}'
ru: Значение JSON по пути (массив ключей): как JSON / как текст.
en: A JSON value by path (an array of keys): as JSON / as text.
ex: SELECT payload #>> '{user,address,city}' FROM events;

@ @> | pg | o
sig: a @> b
ru: Содержит: jsonb содержит jsonb, массив — все элементы другого, диапазон — значение.
en: Contains: jsonb contains jsonb, an array all of another's elements, a range the value.
ex: SELECT * FROM events WHERE payload @> '{"type": "click"}';

@ <@ | pg | o
sig: a <@ b
ru: Содержится в (обратное @>).
en: Is contained in (the reverse of @>).
ex: SELECT * FROM posts WHERE ARRAY['sql'] <@ tags;

@ ? | pg | o
sig: jsonb ? 'key'
ru: Есть ли ключ на верхнем уровне jsonb (?| — любой из, ?& — все).
en: Whether jsonb has the key at the top level (?| any of, ?& all).
ex: SELECT * FROM events WHERE payload ? 'error';

@ && | pg | o
sig: a && b
ru: Пересекаются ли массивы / диапазоны (есть общие элементы).
en: Whether the arrays / ranges overlap (share elements).
ex: SELECT * FROM posts WHERE tags && ARRAY['sql', 'postgres'];

@ ~, ~*, !~, !~* | pg | o
sig: s ~ 'regex' | s ~* 'regex' (без регистра) | !~ — не подходит
ru: Сравнение с регулярным выражением.
en: A regular expression match.
ex: SELECT * FROM customers WHERE phone ~ '^\+7';

@ % | * | o
sig: a % b
ru: Остаток от деления (в Oracle — функция MOD).
en: The remainder of a division (in Oracle, the MOD function).
ex: SELECT * FROM orders WHERE id % 10 = 0;

@ <>, != | * | o
sig: a <> b | a != b
ru: Не равно. С NULL даёт NULL — для «не равно с учётом NULL» есть IS DISTINCT FROM.
en: Not equal. With NULL it's NULL; IS DISTINCT FROM counts NULL in.
ex: SELECT * FROM orders WHERE status <> 'cancelled';

@ = | * | o
sig: a = b
ru: Равно. Сравнение с NULL всегда NULL — используйте IS NULL.
en: Equal. A comparison with NULL is always NULL: use IS NULL.
ex: SELECT * FROM orders WHERE customer_id = 42;

@ <=> | ch | o
sig: a <=> b
ru: Равно, считая NULL = NULL истиной (как IS NOT DISTINCT FROM).
en: Equal, NULL = NULL being true (as IS NOT DISTINCT FROM).
ex: SELECT * FROM a JOIN b ON a.code <=> b.code;

@ / | * | o
sig: a / b
ru: Деление. В PostgreSQL и SQLite целое на целое — целое (7 / 2 = 3): умножьте на 1.0 или приведите к numeric.
en: Division. In PostgreSQL and SQLite integer by integer is an integer (7 / 2 = 3): multiply by 1.0 or cast.
ex: SELECT paid * 1.0 / total AS share FROM orders;

@ (+) | ora | o
sig: WHERE a.id = b.a_id(+)
ru: Старая запись внешнего соединения в Oracle: (+) у таблицы, которой может не быть. Лучше LEFT JOIN.
en: Oracle's old outer join notation: (+) on the side that may be missing. LEFT JOIN is better.
ex: SELECT c.name, o.id FROM customers c, orders o WHERE c.id = o.customer_id(+);
